Google Sheets / AppSheet Web App Connection

Paste your deployed Google Apps Script Web App URL below. When connected, adding/clearing questions automatically clears old rows in Google Sheets and syncs live student submissions.

Google Apps Script (Code.gs) — Fixed Overwrite & Sync Script

Copy this script into Extensions → Apps Script in your Google Sheet and Deploy as Web App (Access: Anyone).

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'getQuestions';
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === 'getQuestions') {
    var qSheet = ss.getSheetByName('Questions') || ss.insertSheet('Questions');
    var lastRow = qSheet.getLastRow();
    if (lastRow < 2) return jsonOut({ status: 'ok', questions: [], settings: getSettingsObj(ss) });
    
    var data = qSheet.getRange(2, 1, lastRow - 1, 7).getValues();
    var questions = data.filter(function(r) { return r[0] !== ''; }).map(function(r, i) {
      return {
        id: 'sq_' + (i + 1),
        text: String(r[0]),
        options: [String(r[1]), String(r[2]), String(r[3]), String(r[4])],
        correctIndex: Number(r[5]) || 0,
        marks: Number(r[6]) || 2.5
      };
    });
    return jsonOut({ status: 'ok', questions: questions, settings: getSettingsObj(ss) });
  }
  return jsonOut({ status: 'ok' });
}

function doPost(e) {
  var payload = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Properly clears old rows before writing new questions so old questions NEVER remain stuck
  if (payload.action === 'saveQuestions') {
    var qSheet = ss.getSheetByName('Questions') || ss.insertSheet('Questions');
    qSheet.getRange(1, 1, 1, 7).setValues([['Question', 'OptionA', 'OptionB', 'OptionC', 'OptionD', 'CorrectIndex', 'Marks']]);
    var lastRow = qSheet.getLastRow();
    if (lastRow >= 2) {
      qSheet.getRange(2, 1, lastRow - 1, 7).clearContent();
    }
    var qs = payload.questions || [];
    if (qs.length > 0) {
      var rows = qs.map(function(q) {
        return [q.text, q.options[0] || '', q.options[1] || '', q.options[2] || '', q.options[3] || '', q.correctIndex, q.marks];
      });
      qSheet.getRange(2, 1, rows.length, 7).setValues(rows);
    }
    if (payload.settings) saveSettingsObj(ss, payload.settings);
    return jsonOut({ status: 'ok', count: qs.length });
  }
  
  if (payload.action === 'submitResult') {
    var rSheet = ss.getSheetByName('Results') || ss.insertSheet('Results');
    if (rSheet.getLastRow() === 0) {
      rSheet.appendRow(['Date', 'QuizID', 'StudentID', 'Name', 'Category', 'Team', 'Batch', 'Score', 'MaxMarks', 'TimeTakenSec', 'SubmittedAt', 'Mode']);
    }
    var r = payload.result;
    rSheet.appendRow([r.dateStr, r.quizId, r.studentId, r.name, r.category, r.team, r.batch, r.score, r.maxMarks, r.timeTakenSeconds, r.submittedAtFormatted, r.mode]);
    return jsonOut({ status: 'ok' });
  }
  return jsonOut({ status: 'unknown_action' });
}

function getSettingsObj(ss) {
  var sSheet = ss.getSheetByName('Settings') || ss.insertSheet('Settings');
  if (sSheet.getLastRow() < 2) return null;
  var row = sSheet.getRange(2, 1, 1, 4).getValues()[0];
  return { title: row[0], durationMinutes: Number(row[1]) || 5, isActive: row[2] === true || row[2] === 'TRUE', quizId: String(row[3] || '') };
}

function saveSettingsObj(ss, s) {
  var sSheet = ss.getSheetByName('Settings') || ss.insertSheet('Settings');
  sSheet.getRange(1, 1, 1, 4).setValues([['Title', 'DurationMinutes', 'IsActive', 'QuizID']]);
  sSheet.getRange(2, 1, 1, 4).setValues([[s.title, s.durationMinutes, s.isActive, s.quizId]]);
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
